const DailyRotateFile = require('winston-daily-rotate-file');
const path = require('path');

const LOG_DIR = process.env.LOG_DIR || './logs';
const LOG_LEVEL = process.env.LOG_LEVEL || 'info';

// Structured JSON format — every field is explicit, no string interpolation
const jsonFormat = winston.format.combine(
  winston.format.timestamp({ format: 'YYYY-MM-DDTHH:mm:ss.SSSZ' }),  // ISO 8601
  winston.format.errors({ stack: true }),
  winston.format.json()
);

// Rotate: new file per day, max 5 MB, keep last 3 files
const fileTransport = new DailyRotateFile({
  dirname:        LOG_DIR,
  filename:       'app-%DATE%.log',
  datePattern:    'YYYY-MM-DD',
  maxSize:        '5m',    // rotate at 5 MB
  maxFiles:       '3',     // keep only 3 files
  format:         jsonFormat,
  zippedArchive:  false,
});

const logger = winston.createLogger({
  level:      LOG_LEVEL,
  format:     jsonFormat,
  transports: [
    fileTransport,
    new winston.transports.Console({
      format: winston.format.combine(
        winston.format.colorize(),
        winston.format.simple()
      ),
    }),
  ],
});

module.exports = logger;
JSsample-app/src/app.js — request logging middleware
const { v4: uuidv4 } = require('uuid');
const logger = require('./logger');

// Request logging middleware: measures response time, logs one line per request
app.use((req, res, next) => {
  const startTime  = Date.now();
  req.requestId    = uuidv4();  // unique ID ties request to log line

  res.on('finish', () => {
    const responseTime = Date.now() - startTime;
    const status       = res.statusCode;

    // Level choice: info=2xx, warn=4xx, error=5xx
    const level =
      status >= 500 ? 'error' :
      status >= 400 ? 'warn'  : 'info';

    logger[level]('HTTP request', {
      timestamp:       new Date().toISOString(),
      endpoint:        req.path,
      method:          req.method,
      status_code:     status,
      response_time_ms: responseTime,
      request_id:      req.requestId,
    });
  });
  next();
});

// Global error handler: captures stack trace
app.use((err, req, res, next) => {
  logger.error('Unhandled error', {
    message:    err.message,
    stack:      err.stack,
    request_id: req.requestId,
    endpoint:   req.path,
  });
  res.status(500).json({ error: 'Internal server error' });
});

  server.close(() => {
    logger.info('HTTP server closed');
    process.exit(0);

const http = require('http');
const args = require('minimist')(process.argv.slice(2));

const RPS        = args.rps      || 2;
const DURATION   = args.duration || 30;   // seconds
const BURST_MODE = args.burst    || false;
const BURST_COUNT= args['burst-count'] || 50;
const BASE_URL   = args.url     || 'http://localhost:3000';

const ENDPOINTS = [
  { path: '/api/users',   method: 'GET'  },
  { path: '/api/orders',  method: 'GET'  },
  { path: '/api/payment', method: 'POST' },
  { path: '/api/search?q=shoes', method: 'GET' },
  { path: '/api/search',  method: 'GET'  },  // missing q → 400
  { path: '/api/health',  method: 'GET'  },
];

let total = 0, success = 0, failure = 0;

function randomEndpoint() {
  return ENDPOINTS[Math.floor(Math.random() * ENDPOINTS.length)];
}

async function sendRequest({ path, method }) {
  return new Promise(resolve => {
    const req = http.request({ hostname: 'localhost', port: 3000, path, method },
      res => {
        total++; res.statusCode < 400 ? success++ : failure++;
        resolve(res.statusCode);
      });
    req.on('error', () => { total++; failure++; resolve(0); });
    req.end();
  });
}

if (BURST_MODE) {
  // Burst: fire BURST_COUNT POST /api/payment requests in rapid succession
  // Used to test alert thresholds in the collector later
  Promise.all(
    Array.from({ length: BURST_COUNT }, () =>
      sendRequest({ path: '/api/payment', method: 'POST' })
    )
  ).then(() =>
    console.log(`Burst done | total=${total} success=${success} failure=${failure}`)
  );
} else {
  const interval = 1000 / RPS;
  const timer = setInterval(async () => {
    await sendRequest(randomEndpoint());
  }, interval);

  setTimeout(() => {
    clearInterval(timer);
    console.log(`\nSummary`);
    console.log(`  Total requests : ${total}`);
    console.log(`  Success        : ${success}`);
    console.log(`  Failure        : ${failure}`);
  }, DURATION * 1000);
}

Pythoncollector/app/watcher.py — tail-f with rotation detection
import asyncio, os, logging
from .parser   import parse_line
from .database import SessionLocal
from .models   import Log, BadLine

BATCH_SIZE = 100  # commit every 100 lines for efficiency
logger = logging.getLogger(__name__)


async def watch_log_file(path: str, poll_interval: float = 0.5):
    """Continuously read new lines from path like `tail -f`.

    Handles two edge cases:
    - Rotation: a new file appears at the same path (inode changes) → reopen from start
    - Truncation: file size shrinks (e.g. logrotate copytruncate) → seek to 0
    """
    pos = 0
    current_inode = None

    while True:
        try:
            stat = os.stat(path)
            # Rotation: inode changed → new file, restart from byte 0
            if current_inode is not None and stat.st_ino != current_inode:
                logger.info("Log rotation detected — reopening file")
                pos = 0
            # Truncation: file is smaller than our position → seek to 0
            if stat.st_size < pos:
                logger.info("Log truncation detected — rewinding to start")
                pos = 0
            current_inode = stat.st_ino

            with open(path, 'r', encoding='utf-8') as f:
                f.seek(pos)
                batch = []
                for raw_line in f:
                    result = parse_line(raw_line)
                    batch.append((result, raw_line))
                    if len(batch) >= BATCH_SIZE:
                        await flush_batch(batch)
                        batch = []
                if batch:
                    await flush_batch(batch)
                pos = f.tell()

        except FileNotFoundError:
            logger.warning(f"Log file not found: {path} — waiting…")
        await asyncio.sleep(poll_interval)


async def flush_batch(batch):
    db = SessionLocal()
    try:
        for result, raw_line in batch:
            if result['ok']:
                entry = result['data']
                # Dedup: skip if same request_id+timestamp already stored
                exists = db.query(Log).filter_by(
                    request_id=entry.request_id,
                    timestamp=entry.timestamp
                ).first()
                if not exists:
                    db.add(Log(**entry.model_dump(), raw_line=raw_line.strip()))
            else:
                db.add(BadLine(raw=raw_line.strip(), reason=result['reason']))
                logger.warning(f"Bad log line: {result['reason']}")
        db.commit()
    finally:
        db.close()
Pythoncollector/app/parser.py — safe JSON parser
import json
from datetime import datetime
from .schemas import LogCreate

REQUIRED_FIELDS = {
    'timestamp', 'level', 'message',
    'endpoint', 'method', 'status_code',
    'response_time_ms', 'request_id',
}
VALID_LEVELS = {'info', 'warn', 'error'}


def parse_line(raw: str) -> dict:
    """Parse one JSON log line. Never raises — returns ok/data or ok=False/reason.

    Returns:
        {'ok': True,  'data': LogCreate(...)}
        {'ok': False, 'reason': 'Missing fields: endpoint, method'}
    """
    raw = raw.strip()
    if not raw:
        return {'ok': False, 'reason': 'empty line'}
    try:
        data = json.loads(raw)
    except json.JSONDecodeError as e:
        return {'ok': False, 'reason': f'Invalid JSON: {e}'}

    missing = REQUIRED_FIELDS - set(data.keys())
    if missing:
        return {'ok': False, 'reason': f'Missing fields: {", ".join(missing)}'}

    if data['level'] not in VALID_LEVELS:
        return {'ok': False, 'reason': f'Unknown level: {data["level"]}'}

    try:
        return {'ok': True, 'data': LogCreate(**data)}
    except Exception as e:
        return {'ok': False, 'reason': f'Validation error: {e}'}
Pythoncollector/app/routes/logs.py — stats endpoints (excerpt)
from fastapi import APIRouter, Depends, Query
from sqlalchemy.orm import Session
from sqlalchemy import func
from ..database import get_db
from ..models   import Log
from ..schemas  import LogQuery, LogsResponse, StatsSummary

router = APIRouter(prefix="/api")


@router.get("/logs", response_model=LogsResponse)
def get_logs(
    level:       str | None = None,
    endpoint:    str | None = None,
    status_code: int | None = None,
    start_time:  str | None = None,
    end_time:    str | None = None,
    search:      str | None = None,  # searches in message field
    page:        int         = Query(1,   ge=1),
    page_size:   int         = Query(50,  ge=1, le=500),
    db: Session              = Depends(get_db),
):
    q = db.query(Log)
    if level:       q = q.filter(Log.level == level)
    if endpoint:    q = q.filter(Log.endpoint == endpoint)
    if status_code: q = q.filter(Log.status_code == status_code)
    if start_time:  q = q.filter(Log.timestamp >= start_time)
    if end_time:    q = q.filter(Log.timestamp <= end_time)
    if search:      q = q.filter(Log.message.contains(search))
    total = q.count()
    items = q.order_by(Log.timestamp.desc()).offset((page-1)*page_size).limit(page_size).all()
    return {"total": total, "page": page, "page_size": page_size, "items": items}


@router.get("/stats/slowest-endpoints")
def slowest_endpoints(db: Session = Depends(get_db)):
    """Top 5 endpoints by average response time + p95 (approximated via percentile_cont)."""
    rows = (
        db.query(
            Log.endpoint,
            func.avg(Log.response_time_ms).label('avg_ms'),
            func.count().label('count'),
        )
        .group_by(Log.endpoint)
        .order_by(func.avg(Log.response_time_ms).desc())
        .limit(5)
        .all()
    )
    return [
        {"endpoint": r.endpoint, "avg_ms": round(r.avg_ms, 1), "count": r.count}
        for r in rows
    ]

— Alert System

Three Alert Rules

🔴 Error Spike
Fires when more than 10 error-level logs appear in the last 5 minutes. Indicates a systematic failure in an endpoint or downstream service.
threshold: 10 errors / 5 min
cooldown: 10 min · auto-resolves when count drops
🟡 Slow Response
Fires when the average response_time_ms across all endpoints exceeds 1000 ms in the last 5 minutes. Points to a slow DB query or memory pressure.
threshold: avg > 1000 ms / 5 min
cooldown: 10 min · auto-resolves when avg normalises
🟠 High Error Rate
Fires when errors exceed 30% of all requests in the last 5 minutes. A burst of POST /api/payment failures is a typical trigger.
threshold: error rate > 30%
cooldown: 10 min · auto-resolves when rate drops
Pythoncollector/app/alert_checker.py — background task
import asyncio, json, logging
from datetime import datetime, timedelta
from .database import SessionLocal
from .models   import Log, Alert

logger = logging.getLogger(__name__)
ALERT_LOG = "alerts.log"

# Loaded from alert_rules.json at startup
RULES = [
  { "name": "error_spike",  "window_minutes": 5, "threshold": 10,   "metric": "error_count"  },
  { "name": "slow_response", "window_minutes": 5, "threshold": 1000, "metric": "avg_response_ms"},
  { "name": "high_error_rate","window_minutes": 5, "threshold": 30,   "metric": "error_rate_pct" },
]
COOLDOWN_MINUTES = 10  # do not re-fire the same rule within this window


def compute_metric(db, rule: dict) -> float:
    since = datetime.utcnow() - timedelta(minutes=rule["window_minutes"])
    rows  = db.query(Log).filter(Log.timestamp >= since.isoformat()).all()
    if not rows: return 0.0
    if rule["metric"] == "error_count":
        return sum(1 for r in rows if r.level == "error")
    if rule["metric"] == "avg_response_ms":
        return sum(r.response_time_ms for r in rows) / len(rows)
    if rule["metric"] == "error_rate_pct":
        errors = sum(1 for r in rows if r.level == "error")
        return (errors / len(rows)) * 100
    return 0.0


async def check_alerts():
    """Run every 30 seconds. For each rule: compute metric, fire or resolve alert."""
    while True:
        await asyncio.sleep(30)
        db = SessionLocal()
        try:
            for rule in RULES:
                value = compute_metric(db, rule)
                triggered = value > rule["threshold"]

                active_alert = db.query(Alert).filter_by(
                    rule_name=rule["name"], status="active"
                ).first()

                if triggered and not active_alert:
                    # Cooldown: skip if the same rule fired within 10 minutes
                    cooldown_since = datetime.utcnow() - timedelta(minutes=COOLDOWN_MINUTES)
                    recent = db.query(Alert).filter(
                        Alert.rule_name == rule["name"],
                        Alert.triggered_at >= cooldown_since
                    ).first()
                    if not recent:
                        msg = f"{rule['name']}: {value:.1f} exceeded {rule['threshold']}"
                        alert = Alert(rule_name=rule["name"], message=msg,
                                       value=value, threshold=rule["threshold"],
                                       status="active")
                        db.add(alert); db.commit()
                        # Also write to alerts.log
                        with open(ALERT_LOG, 'a') as f:
                            f.write(json.dumps({
                                "triggered_at": datetime.utcnow().isoformat(),
                                "rule": rule["name"], "value": value, "message": msg
                            }) + "\n")
                        logger.warning(f"ALERT FIRED: {msg}")

                elif not triggered and active_alert:
                    # Condition cleared → auto-resolve
                    active_alert.status = "resolved"
                    active_alert.resolved_at = datetime.utcnow()
                    db.commit()
                    logger.info(f"Alert resolved: {rule['name']}")
        finally:
            db.close()

/* api.js — all collector requests go through these helpers.
   On network failure, sets the connection-lost banner instead of crashing. */
import { API_BASE } from './config.js';

let _connectionOk = true;

async function apiFetch(path, opts = {}) {
  try {
    const res = await fetch(API_BASE + path, opts);
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    _connectionOk = true;
    return res.json();
  } catch (err) {
    _connectionOk = false;
    setConnectionStatus(false);   // shows yellow "Connection lost" banner
    throw err;
  }
}

export const fetchLogs    = (params) => apiFetch(`/api/logs?${new URLSearchParams(params)}`);
export const fetchSummary = (w = 60)  => apiFetch(`/api/stats/summary?window_minutes=${w}`);
export const fetchEph     = ()        => apiFetch('/api/stats/errors-per-hour');
export const fetchSlowest = ()        => apiFetch('/api/stats/slowest-endpoints');
export const fetchAlerts  = ()        => apiFetch('/api/alerts?status=active');
export const ackAlert     = (id)      => apiFetch(`/api/alerts/${id}/acknowledge`, { method: 'POST' });

function setConnectionStatus(ok) {
  const el = document.getElementById('conn-status');
  if (!el) return;
  el.className = ok ? 'conn-status conn-ok' : 'conn-status conn-warn';
  el.innerHTML = ok
    ? '<div class="conn-dot"></div> Connected'
    : '<div class="conn-dot"></div> Connection lost, retrying…';
}

          docker push ghcr.io/${{ github.repository }}/collector:latest
          docker push ghcr.io/${{ github.repository }}/dashboard:latest

      - name: Deploy to EC2 via SSH
        uses: appleboy/ssh-action@v1
        with:
          host:     ${{ secrets.EC2_HOST }}

            docker compose pull
            docker compose up -d

— AWS

EC2 Deployment Steps (DEPLOYMENT.md)

1
Launch EC2 instance
Go to AWS Console → EC2 → Launch Instance. Choose Ubuntu 22.04 LTS, instance type t2.micro (free tier) or t3.micro. Create a new key pair (.pem file) and download it — you'll need it for SSH.
2
Configure Security Group — open only what's needed
Add inbound rules: SSH port 22 from your IP only, HTTP port 80 from anywhere (dashboard), and optionally port 8080 for the dashboard if you change it. Do not open port 3000 (sample app) or 8000 (collector) to the public — they talk to each other over the internal Docker network, not the internet.
⚠ Opening the collector API publicly lets anyone query or corrupt your log data.
3
Install Docker on the instance
SSH in with ssh -i key.pem ubuntu@<ip>, then run:
sudo apt-get update && sudo apt-get install -y docker.io docker-compose-plugin
sudo usermod -aG docker ubuntu && newgrp docker
4
Clone repo and create .env
git clone https://github.com/you/log-monitor.git && cd log-monitor
Copy .env.example to .env and fill in your values. Run docker compose up -d --build. Visit http://<ec2-ip> to see the dashboard.
5
Watch logs and roll back
Tail all logs: docker compose logs -f. One service: docker compose logs -f collector. Roll back: docker compose down && docker pull ghcr.io/you/collector:v1.0.0 && docker compose up -d. Tag your releases so you have a previous version to pull.
6
Cost and billing safety
A t2.micro runs ~$0.012/hour → roughly $8–9/month if left on 24/7. Set a AWS Billing Alert (Budget → Create Budget → $10 threshold) so you get an email before it surprises you. Stop the instance when not in use: aws ec2 stop-instances --instance-ids i-xxx. Stopped instances don't charge for compute (only the EBS volume, ~$0.10/GB/month).
⚠ Always check the Billing Dashboard after leaving an instance running overnight.

— Design Decisions

Why These Choices?

Why SQLite and not PostgreSQL?
SQLite is zero-setup — the DB is a single file, no service to run, perfect for a single-node project. At this scale (thousands of log rows/day) it's fast enough. For production with multiple collector replicas, switch to PostgreSQL.
Why file tailing instead of a message queue?
File tailing (tail -f) requires zero extra infrastructure — the log file already exists. A message queue (Kafka, Redis Pub/Sub) adds ops overhead. At scale, switch to Fluent Bit → Kafka for reliable, back-pressure-aware delivery.
What would change for production?
Replace SQLite with PostgreSQL, add an index on (timestamp, level). Replace the watcher with Fluent Bit or Logstash. Replace Chart.js with Grafana + Prometheus. Add authentication on the collector API. Use ECS or Kubernetes instead of a single EC2.
Why plain HTML/CSS for the dashboard?
No build tooling needed — the dashboard is a few static files served by nginx. No npm install, no bundler, one Dockerfile line. The Chart.js CDN script is the only dependency. For a team project, React + Vite would be a better choice.
